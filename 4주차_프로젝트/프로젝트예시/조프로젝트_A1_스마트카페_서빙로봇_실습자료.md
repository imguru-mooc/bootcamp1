## 1. 프로젝트 개요

### 1.1 시나리오

손님이 테이블에 놓인 QR을 폰으로 스캔하면 주문 페이지가 열립니다.
주문을 넣으면 주방 대시보드에 실시간으로 접수되고, 서빙 로봇이
해당 테이블까지 주행해 LCD에 주문 내역을 표시하고 부저로 도착을
알립니다. 손님이 다가오면 로봇 거치 폰의 얼굴 감지가 수령을 확인하고,
로봇은 홈 위치로 복귀해 다음 주문을 처리합니다.

### 1.2 학습 목표

1. 주문 큐(FIFO)를 관리하는 상태 기반 서비스 노드를 설계할 수 있다.
2. 시간 기반 주행 시퀀스와 연속 명령 발행(워치독)을 구현할 수 있다.
3. 도착 알림(LCD·부저)을 엣지 명령으로 분리 설계할 수 있다.
4. 브라우저 얼굴 감지 결과를 ROS 2 이벤트로 연결할 수 있다.
5. 4개 서브시스템을 토픽 인터페이스로 통합하고 E2E 테스트할 수 있다.

### 1.3 시스템 구조

```
〔손님 폰〕 주문 웹앱 (QR → ?table=N)          〔주방 폰/PC〕 대시보드
     │ /order (JSON, 엣지)                        ▲ /order_state (JSON, 엣지)
     ▼                                            │
   wss://<Pi IP>:9090 ─── rosbridge_server ───────┘
     │
     ▼
〔Pi〕 order_manager ──/goal_table──▶ table_driver ──/cmd_vel(연속)──▶ 터틀봇
         ▲                                │
         │ /delivery_done                 │ /arrived
         │                                ▼
       delivery_coordinator ──/lcd_text·/buzzer(엣지)──▶ LCD·부저
         ▲        │
         │        └─/return_home──▶ table_driver
         │ /face_detected
〔로봇 거치 폰〕 얼굴 감지 웹앱 (MediaPipe Face Detector)
```

**order_manager.py — 주문 큐 관리**

```python
#!/usr/bin/env python3
"""order_manager: 주문 접수 → FIFO 큐 관리 → 배달 지시"""
import json
import rclpy
from rclpy.node import Node
from std_msgs.msg import String


class OrderManager(Node):
    def __init__(self):
        super().__init__('order_manager')
        # 주문 큐: [{'order_id':..., 'table':..., 'items':[...]}]
        self.queue = []
        self.current = None          # 배달 중인 주문
        self.sub_order = self.create_subscription(
            String, '/order', self.on_order, 10)
        self.sub_done = self.create_subscription(
            String, '/delivery_done', self.on_done, 10)
        self.pub_goal = self.create_publisher(String, '/goal_table', 10)
        self.pub_state = self.create_publisher(String, '/order_state', 10)
        self.get_logger().info('order_manager 시작')

    def on_order(self, msg):
        """웹 주문 수신 (JSON): {"order_id":"...","table":2,"items":["아메리카노"]}"""
        try:
            order = json.loads(msg.data)
        except json.JSONDecodeError:
            self.get_logger().warn(f'잘못된 주문 형식: {msg.data}')
            return
        order['status'] = '접수'
        self.queue.append(order)
        self.broadcast_state()
        self.try_dispatch()

    def on_done(self, msg):
        """coordinator가 배달 완료를 알림 (order_id)"""
        if self.current and self.current['order_id'] == msg.data:
            self.current['status'] = '완료'
            self.broadcast_state()
            self.current = None
            self.try_dispatch()

    def try_dispatch(self):
        """배달 중이 아니고 큐에 주문이 있으면 다음 주문 출발"""
        if self.current is not None or not self.queue:
            return
        self.current = self.queue.pop(0)      # FIFO
        self.current['status'] = '배달중'
        goal = String()
        goal.data = json.dumps(self.current, ensure_ascii=False)
        self.pub_goal.publish(goal)
        self.broadcast_state()

    def broadcast_state(self):
        """대시보드용 전체 상태 발행 (엣지: 변화 시에만 호출됨)"""
        state = {
            'current': self.current,
            'queue': self.queue,
        }
        msg = String()
        msg.data = json.dumps(state, ensure_ascii=False)
        self.pub_state.publish(msg)


def main():
    rclpy.init()
    node = OrderManager()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
```

**table_driver.py — 시간 기반 주행 시퀀스**

```python
#!/usr/bin/env python3
"""table_driver: 목표 테이블까지 시간 기반 주행 시퀀스 실행"""
import json
import rclpy
from rclpy.node import Node
from std_msgs.msg import String
from geometry_msgs.msg import Twist

# ── 테이블별 주행 시퀀스: (linear.x, angular.z, 지속 초)
# TODO(조별 작업): 실측하여 값을 보정하고 테이블을 추가하세요.
ROUTES = {
    1: [(0.10, 0.0, 3.0), (0.0, 0.5, 3.1), (0.10, 0.0, 2.0)],
    2: [(0.10, 0.0, 5.0), (0.0, -0.5, 3.1), (0.10, 0.0, 2.0)],
}
HOME = 'home'   # 복귀 명령 키워드


def reverse_route(route):
    """왕복용: 시퀀스를 역순·역방향으로 변환"""
    return [(-lx, -az, t) for (lx, az, t) in reversed(route)]


class TableDriver(Node):
    def __init__(self):
        super().__init__('table_driver')
        self.sub_goal = self.create_subscription(
            String, '/goal_table', self.on_goal, 10)
        self.sub_return = self.create_subscription(
            String, '/return_home', self.on_return, 10)
        self.pub_vel = self.create_publisher(Twist, '/cmd_vel', 10)
        self.pub_arrived = self.create_publisher(String, '/arrived', 10)
        # 시퀀스 실행 상태
        self.route = []            # 남은 (lx, az, 남은시간) 목록
        self.order_json = ''
        self.last_table = None
        # 연속 발행 루프 (10Hz, 워치독 충족: 대기 중에도 정지 명령 발행)
        self.timer = self.create_timer(0.1, self.tick)
        self.get_logger().info('table_driver 시작')

    def on_goal(self, msg):
        order = json.loads(msg.data)
        table = int(order['table'])
        if table not in ROUTES:
            self.get_logger().warn(f'미정의 테이블: {table}')
            return
        self.route = [list(step) for step in ROUTES[table]]
        self.order_json = msg.data
        self.last_table = table
        self.get_logger().info(f'{table}번 테이블 출발')

    def on_return(self, msg):
        """coordinator의 복귀 지시 → 마지막 경로 역주행"""
        if self.last_table is not None:
            self.route = [list(s) for s in reverse_route(ROUTES[self.last_table])]
            self.get_logger().info('복귀 시작')

    def tick(self):
        """10Hz: 시퀀스 소비하며 연속 발행. 시퀀스 없으면 정지 발행."""
        cmd = Twist()
        if self.route:
            lx, az, remain = self.route[0]
            cmd.linear.x = float(lx)
            cmd.angular.z = float(az)
            self.route[0][2] = remain - 0.1
            if self.route[0][2] <= 0:
                self.route.pop(0)
                if not self.route and self.order_json:
                    arrived = String()
                    arrived.data = self.order_json
                    self.pub_arrived.publish(arrived)
                    self.order_json = ''
        self.pub_vel.publish(cmd)   # 시퀀스 없으면 0,0 → 안전 정지 유지


def main():
    rclpy.init()
    node = TableDriver()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
```

**delivery_coordinator.py — 도착·수령·복귀 조정**

```python
#!/usr/bin/env python3
"""delivery_coordinator: 도착 → LCD·부저 → 얼굴 확인 → 복귀·완료 처리"""
import json
import rclpy
from rclpy.node import Node
from std_msgs.msg import String, Bool

PICKUP_TIMEOUT = 30.0     # 수령 대기 제한 (초)


class DeliveryCoordinator(Node):
    def __init__(self):
        super().__init__('delivery_coordinator')
        self.sub_arrived = self.create_subscription(
            String, '/arrived', self.on_arrived, 10)
        self.sub_face = self.create_subscription(
            Bool, '/face_detected', self.on_face, 10)
        self.pub_lcd = self.create_publisher(String, '/lcd_text', 10)
        self.pub_buzzer = self.create_publisher(Bool, '/buzzer', 10)
        self.pub_return = self.create_publisher(String, '/return_home', 10)
        self.pub_done = self.create_publisher(String, '/delivery_done', 10)
        self.waiting = None        # 수령 대기 중인 주문 (dict)
        self.wait_timer = None
        self.get_logger().info('delivery_coordinator 시작')

    def on_arrived(self, msg):
        order = json.loads(msg.data)
        self.waiting = order
        # LCD·부저는 엣지 명령: 도착 시 1회만 발행
        lcd = String()
        items = ','.join(order.get('items', []))
        lcd.data = f"T{order['table']} {items}"
        self.pub_lcd.publish(lcd)
        buz = Bool()
        buz.data = True
        self.pub_buzzer.publish(buz)
        # 수령 타임아웃 타이머 시작
        self.wait_timer = self.create_timer(PICKUP_TIMEOUT, self.on_timeout)
        self.get_logger().info(f"T{order['table']} 도착, 수령 대기")

    def on_face(self, msg):
        """로봇 거치 폰의 얼굴 감지 (True 수신 시 수령으로 간주)"""
        if self.waiting and msg.data:
            self.finish('수령 확인')

    def on_timeout(self):
        if self.waiting:
            self.finish('타임아웃 복귀')

    def finish(self, reason):
        order = self.waiting
        self.waiting = None
        if self.wait_timer:
            self.wait_timer.cancel()
            self.wait_timer = None
        lcd = String()
        lcd.data = f"{reason}"
        self.pub_lcd.publish(lcd)
        ret = String()
        ret.data = 'home'
        self.pub_return.publish(ret)
        done = String()
        done.data = order['order_id']
        self.pub_done.publish(done)
        self.get_logger().info(f"{order['order_id']} {reason}")


def main():
    rclpy.init()
    node = DeliveryCoordinator()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
```

💡 **스타터 코드는 출발점입니다.** 그대로 제출하면 기능 점수의 절반만
인정됩니다. Should/Could 구현, 경로 실측 보정, 예외 처리 보강,
S2 타임아웃의 대시보드 연동 등이 조의 몫입니다.

```python
#!/usr/bin/env python3
"""buzzer_node: /buzzer 수신 시 도착 알림음 (0.3초 x 3회)"""
import rclpy
from rclpy.node import Node
from std_msgs.msg import Bool
from gpiozero import Buzzer   # export GPIOZERO_PIN_FACTORY=lgpio

BUZZER_PIN = 17               # Freenove 배선에 맞게 수정


class BuzzerNode(Node):
    def __init__(self):
        super().__init__('buzzer_node')
        self.buzzer = Buzzer(BUZZER_PIN)
        self.sub = self.create_subscription(Bool, '/buzzer', self.on_msg, 10)
        self.get_logger().info('buzzer_node 시작')

    def on_msg(self, msg):
        if msg.data:
            self.buzzer.beep(on_time=0.3, off_time=0.2, n=3, background=True)


def main():
    rclpy.init()
    node = BuzzerNode()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
```
